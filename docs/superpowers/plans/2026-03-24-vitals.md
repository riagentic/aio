# Vital Signs Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use
> superpowers:subagent-driven-development (recommended) or
> superpowers:executing-plans to implement this plan task-by-task. Steps use
> checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a three-layer diagnostic system (render, transport, loop) that
detects UI freezes, identifies root cause, and provides actionable fix hints.

**Architecture:** Three probes (RenderProbe, TransportProbe, LoopProbe) feed
into a hint engine that correlates freeze signals across client/server boundary.
Server acts as watchdog (detects frozen clients in real-time), client provides
post-mortem analysis on recovery. Dual-sided detection with cascade
directionality.

**Tech Stack:** TypeScript, Deno 2.6+, WebSocket/UDS, `@std/assert` for tests

**Spec:** `docs/superpowers/specs/2026-03-24-vitals-design.md`

```typescript
// src/vitals/hints.ts
import type {
  ProbeTimeline,
  VitalHint,
  VitalLayer,
  VitalsSnapshot,
  VitalThresholds,
} from "./types.ts";

/** Classify severity based on evidence */
export function classifySeverity(
  probeCount: number,
  hasDirectMeasurement: boolean,
  hasCrossProbeCorrelation: boolean,
): "likely" | "possible" | "speculative" {
  if (probeCount >= 2 && hasDirectMeasurement && hasCrossProbeCorrelation) {
    return "likely";
  }
  if (probeCount >= 1 && hasDirectMeasurement) return "possible";
  return "speculative";
}

/** Find which probe degraded first */
export function detectCascadeOrigin(timelines: ProbeTimeline[]): VitalLayer {
  const withDeg = timelines.filter((t) => t.firstDegradedAt !== null);
  if (withDeg.length === 0) return "render"; // fallback
  withDeg.sort((a, b) => a.firstDegradedAt! - b.firstDegradedAt!);
  return withDeg[0].probe;
}

/** Evaluate all hint rules against a snapshot. Returns the first matching hint, or null. */
export function evaluateHints(
  snap: VitalsSnapshot,
  thresholds: VitalThresholds,
): VitalHint | null {
  // Rule 6: Visibility filter — discard if window is hidden
  if (snap.render.status === "frozen" && snap.render.visible === false) {
    return null;
  }

  // Rule 5: Recovery death spiral — repeated freeze/recover
  if (
    snap.render.status === "recovered" &&
    (snap.render.previousFreezeCount ?? 0) > 1
  ) {
    return {
      cause:
        `Repeated freeze-recover cycle (${snap.render.previousFreezeCount} times)`,
      evidence: [`${snap.render.previousFreezeCount} freezes in last 30s`],
      suggestion:
        "If backpressure is off, enable it. If on, check for expensive onVitalAlert handlers or post-recovery reconciliation.",
      severity: "likely",
    };
  }

  // Rule 1: Slow reduce → render freeze
  // Uses degraded threshold (100ms default) as budget — not frozen (2000ms)
  if (
    snap.render.status === "frozen" &&
    snap.loop.lastReduceTime > thresholds.loop.degraded &&
    snap.loop.lastReduceAction === snap.render.lastActionBefore
  ) {
    return {
      cause:
        `Reducer for '${snap.loop.lastReduceFeature}/${snap.loop.lastReduceAction}' took ${
          Math.round(snap.loop.lastReduceTime)
        }ms`,
      evidence: [
        `reduce took ${
          Math.round(snap.loop.lastReduceTime)
        }ms (budget: ${thresholds.loop.degraded}ms)`,
        `render frozen for ${Math.round(snap.render.measured)}ms`,
      ],
      suggestion: "Optimize the reduce, or split into smaller actions.",
      severity: classifySeverity(2, true, true),
    };
  }

  // Rule 2: Queue saturation
  if (
    snap.loop.queueDepth > thresholds.queue.frozen &&
    snap.loop.firstDegradedAt !== null &&
    (snap.transport.firstDegradedAt === null ||
      snap.loop.firstDegradedAt < snap.transport.firstDegradedAt)
  ) {
    return {
      cause:
        `Dispatch queue backed up to ${snap.loop.queueDepth} actions, drain rate: ${snap.loop.drainRate}/s`,
      evidence: [
        `queue depth: ${snap.loop.queueDepth} (threshold: ${thresholds.queue.frozen})`,
        `top contributor: ${snap.loop.lastReduceFeature}/${snap.loop.lastReduceAction}`,
      ],
      suggestion: "Debounce rapid-fire dispatches, or batch related actions.",
      severity: classifySeverity(2, true, true),
    };
  }

  // Rule 3: Transport stall (both sides healthy)
  if (
    snap.transport.status === "frozen" &&
    snap.render.status === "healthy" &&
    snap.loop.status === "healthy"
  ) {
    return {
      cause: `Network connection stalled. No pong in ${
        Math.round(snap.transport.measured)
      }ms`,
      evidence: [
        `transport frozen: ${
          Math.round(snap.transport.measured)
        }ms (threshold: ${thresholds.transport.frozen}ms)`,
      ],
      suggestion: "Check network stability. Auto-reconnect will trigger.",
      severity: classifySeverity(1, true, false),
    };
  }

  // Rule 4: Client-only freeze (non-AIO code)
  if (
    snap.render.status === "frozen" &&
    snap.transport.status === "healthy" &&
    snap.loop.status === "healthy" &&
    snap.render.lastActionBefore === null
  ) {
    return {
      cause: "Main thread blocked by non-AIO code",
      evidence: [
        `render frozen: ${Math.round(snap.render.measured)}ms`,
        "no recent AIO action before freeze",
      ],
      suggestion:
        "Check third-party libraries, large DOM operations, synchronous I/O.",
      severity: "possible",
    };
  }

  // No matching rule
  return null;
}
```

```typescript
// src/vitals/loop-probe.ts
import type { LoopVitals, VitalStatus, VitalThresholds } from "./types.ts";
import type { PerfTiming } from "../dispatch.ts";

const P95_WINDOW = 100;
const DRAIN_WINDOW_MS = 5000;

export type LoopProbeAPI = {
  onPerf: (timing: PerfTiming) => void;
  updateQueueDepth: (depth: number) => void;
  updateEffectBacklog: (count: number) => void;
  updateCircuitBreakers: (names: string[]) => void;
  getVitals: () => LoopVitals;
  getStatus: () => VitalStatus;
  getFirstDegradedAt: () => number | null;
  reset: () => void;
};

export function createLoopProbe(thresholds: VitalThresholds): LoopProbeAPI {
  let queueDepth = 0;
  let effectBacklog = 0;
  let lastReduceTime = 0;
  let lastReduceAction = "";
  let lastReduceFeature = "";
  let circuitBreakers: string[] = [];
  let firstDegradedAt: number | null = null;

  // p95 tracking
  const reduceTimes: number[] = [];

  // drain rate tracking
  const actionTimestamps: number[] = [];

  function computeP95(): number {
    if (reduceTimes.length === 0) return 0;
    const sorted = [...reduceTimes].sort((a, b) => a - b);
    const idx = Math.ceil(sorted.length * 0.95) - 1;
    return sorted[Math.max(0, idx)];
  }

  function computeDrainRate(): number {
    const now = Date.now();
    const cutoff = now - DRAIN_WINDOW_MS;
    // Remove old timestamps
    while (actionTimestamps.length > 0 && actionTimestamps[0] < cutoff) {
      actionTimestamps.shift();
    }
    if (actionTimestamps.length === 0) return 0;
    const elapsed = (now - actionTimestamps[0]) / 1000;
    return elapsed > 0
      ? Math.round(actionTimestamps.length / elapsed)
      : actionTimestamps.length;
  }

  function getStatus(): VitalStatus {
    // Check queue depth thresholds
    const qt = thresholds.queue;
    if (queueDepth >= qt.frozen) return "frozen";
    if (queueDepth >= qt.warning) return "warning";
    if (queueDepth >= qt.degraded) return "degraded";

    // Check reduce time thresholds
    const lt = thresholds.loop;
    if (lastReduceTime >= lt.frozen) return "frozen";
    if (lastReduceTime >= lt.warning) return "warning";
    if (lastReduceTime >= lt.degraded) return "degraded";

    return "healthy";
  }

  return {
    onPerf(timing: PerfTiming) {
      lastReduceTime = timing.reduce;
      lastReduceAction = timing.actionType;
      // Extract feature name from "feature/ACTION" or "feature:ACTION" patterns
      const sep = timing.actionType.indexOf("/");
      lastReduceFeature = sep > 0
        ? timing.actionType.slice(0, sep)
        : timing.actionType.split(":")[0] ?? "";

      // Track for p95
      reduceTimes.push(timing.reduce);
      if (reduceTimes.length > P95_WINDOW) reduceTimes.shift();

      // Track for drain rate
      actionTimestamps.push(Date.now());

      // Track first degraded
      const status = getStatus();
      if (status !== "healthy" && firstDegradedAt === null) {
        firstDegradedAt = Date.now();
      } else if (status === "healthy") {
        firstDegradedAt = null;
      }
    },

    updateQueueDepth(depth: number) {
      queueDepth = depth;
      const status = getStatus();
      if (status !== "healthy" && firstDegradedAt === null) {
        firstDegradedAt = Date.now();
      } else if (status === "healthy") {
        firstDegradedAt = null;
      }
    },

    updateEffectBacklog(count: number) {
      effectBacklog = count;
    },

    updateCircuitBreakers(names: string[]) {
      circuitBreakers = names;
    },

    getVitals(): LoopVitals {
      return {
        queueDepth,
        drainRate: computeDrainRate(),
        lastReduceTime,
        lastReduceAction,
        lastReduceFeature,
        p95ReduceTime: computeP95(),
        effectBacklog,
        circuitBreakers,
      };
    },

    getStatus,
    getFirstDegradedAt: () => firstDegradedAt,

    reset() {
      queueDepth = 0;
      effectBacklog = 0;
      lastReduceTime = 0;
      lastReduceAction = "";
      lastReduceFeature = "";
      circuitBreakers = [];
      firstDegradedAt = null;
      reduceTimes.length = 0;
      actionTimestamps.length = 0;
    },
  };
}
```

```typescript
// src/vitals/render-probe.ts
import type {
  RenderFreezeReport,
  VitalStatus,
  VitalThresholds,
} from "./types.ts";

export type RenderProbeConfig = {
  thresholds: VitalThresholds;
  interval: number;
  /** If true, no setTimeout loop — call tick() manually. For testing. */
  manualTick?: boolean;
  /** Called when status changes */
  onStatusChange?: (
    status: VitalStatus,
    report: RenderFreezeReport | null,
  ) => void;
};

export type RenderProbeAPI = {
  getStatus: () => VitalStatus;
  getFirstDegradedAt: () => number | null;
  getLastAction: () => string | null;
  getLastFeature: () => string | null;
  getUnprocessedDeltas: () => number;
  getPreviousFreezeCount: () => number;
  getMeasured: () => number;
  recordAction: (actionType: string, feature: string) => void;
  recordDelta: () => void;
  clearDeltas: () => void;
  /** Manual tick for testing: simulate elapsed ms since last tick. Returns freeze report if frozen. */
  tick: (elapsedMs: number) => RenderFreezeReport | null;
  destroy: () => void;
};

export function createRenderProbe(config: RenderProbeConfig): RenderProbeAPI {
  const { thresholds, interval, manualTick, onStatusChange } = config;
  const rt = thresholds.render;

  let status: VitalStatus = "healthy";
  let firstDegradedAt: number | null = null;
  let lastAction: string | null = null;
  let lastFeature: string | null = null;
  let unprocessedDeltas = 0;
  let previousFreezeCount = 0;
  let lastTickAt = Date.now();
  let measured = 0;
  let timer: ReturnType<typeof setTimeout> | null = null;
  let destroyed = false;

  // Freeze count reset timer — counts freezes in 30s window
  let freezeCountResetTimer: ReturnType<typeof setTimeout> | null = null;

  function setStatus(
    next: VitalStatus,
    report: RenderFreezeReport | null = null,
  ) {
    if (next === status) return;
    const prev = status;
    status = next;

    // Track first degradation
    if (
      next !== "healthy" && next !== "recovered" && firstDegradedAt === null
    ) {
      firstDegradedAt = Date.now();
    } else if (next === "healthy") {
      firstDegradedAt = null;
    }

    // Track recovery from frozen
    if (prev === "frozen" && (next === "recovered" || next === "healthy")) {
      previousFreezeCount++;
      // Reset freeze count after 30s
      if (freezeCountResetTimer) clearTimeout(freezeCountResetTimer);
      freezeCountResetTimer = setTimeout(() => {
        previousFreezeCount = 0;
      }, 30_000);
    }

    onStatusChange?.(next, report);
  }

  function evaluateTick(elapsedMs: number): RenderFreezeReport | null {
    measured = elapsedMs;
    let report: RenderFreezeReport | null = null;

    if (elapsedMs >= rt.frozen) {
      report = {
        frozenFor: elapsedMs,
        lastActionBefore: lastAction,
        lastFeature,
        unprocessedDeltas,
        memoryBefore: undefined,
        memoryAfter: undefined,
      };
      setStatus("frozen", report);
    } else if (elapsedMs >= rt.warning) {
      setStatus("warning");
    } else if (elapsedMs >= rt.degraded) {
      setStatus("degraded");
    } else {
      // Healthy tick — if we were frozen, transition to recovered
      if (status === "frozen") {
        setStatus("recovered");
      } else if (status !== "recovered") {
        setStatus("healthy");
      }
    }

    return report;
  }

  function scheduleNext() {
    if (destroyed || manualTick) return;
    timer = setTimeout(() => {
      if (destroyed) return;
      const now = Date.now();
      const elapsed = now - lastTickAt;
      lastTickAt = now;
      evaluateTick(elapsed - interval); // drift = elapsed - expected
      scheduleNext();
    }, interval);
  }

  // Auto-start if not manual
  if (!manualTick) {
    scheduleNext();
  }

  return {
    getStatus: () => status,
    getFirstDegradedAt: () => firstDegradedAt,
    getLastAction: () => lastAction,
    getLastFeature: () => lastFeature,
    getUnprocessedDeltas: () => unprocessedDeltas,
    getPreviousFreezeCount: () => previousFreezeCount,
    getMeasured: () => measured,

    recordAction(actionType: string, feature: string) {
      lastAction = actionType;
      lastFeature = feature;
    },

    recordDelta() {
      unprocessedDeltas++;
    },

    clearDeltas() {
      unprocessedDeltas = 0;
    },

    tick(elapsedMs: number): RenderFreezeReport | null {
      lastTickAt = Date.now();
      return evaluateTick(elapsedMs);
    },

    destroy() {
      destroyed = true;
      if (timer) clearTimeout(timer);
      if (freezeCountResetTimer) clearTimeout(freezeCountResetTimer);
      timer = null;
      status = "healthy";
      firstDegradedAt = null;
      measured = 0;
    },
  };
}
```

```typescript
import type {
  ClientLiveness,
  LoopVitals,
  VitalStatus,
  VitalThresholds,
} from "./types.ts";

// ─── Wire Protocol Types ──────────────────────────────────────────────────────

export type VitalsPing = { type: "__vitals:ping"; t1: number };
export type VitalsPong = {
  type: "__vitals:pong";
  t1: number;
  t2: number;
  loop: LoopVitals | null;
};

// ─── Client-Side Probe ────────────────────────────────────────────────────────

export type TransportProbeClientConfig = {
  thresholds: VitalThresholds;
  interval: number;
  onStatusChange?: (status: VitalStatus) => void;
};

export type TransportProbeClientAPI = {
  createPing: () => VitalsPing;
  processPong: (
    pong: { t1: number; t2: number; loop: LoopVitals | null },
  ) => void;
  checkLiveness: (lastPongAt: number) => void;
  getStatus: () => VitalStatus;
  getRTT: () => number;
  getFirstDegradedAt: () => number | null;
  getLastLoop: () => LoopVitals | null;
  destroy: () => void;
};

export function createTransportProbeClient(
  config: TransportProbeClientConfig,
): TransportProbeClientAPI {
  const tt = config.thresholds.transport;
  let status: VitalStatus = "healthy";
  let rtt = 0;
  let firstDegradedAt: number | null = null;
  let lastPongAt = Date.now();
  let lastLoop: LoopVitals | null = null;

  function updateStatus(next: VitalStatus) {
    if (next === status) return;
    status = next;
    if (
      next !== "healthy" && next !== "recovered" && firstDegradedAt === null
    ) {
      firstDegradedAt = Date.now();
    } else if (next === "healthy") {
      firstDegradedAt = null;
    }
    config.onStatusChange?.(next);
  }

  function evaluateRTT(measuredRTT: number) {
    rtt = measuredRTT;
    if (measuredRTT >= tt.frozen) updateStatus("frozen");
    else if (measuredRTT >= tt.warning) updateStatus("warning");
    else if (measuredRTT >= tt.degraded) updateStatus("degraded");
    else updateStatus("healthy");
  }

  return {
    createPing(): VitalsPing {
      return { type: "__vitals:ping", t1: Date.now() };
    },

    processPong(pong) {
      const now = Date.now();
      lastPongAt = now;
      const measuredRTT = now - pong.t1;
      lastLoop = pong.loop;
      evaluateRTT(measuredRTT);
    },

    checkLiveness(lastPong: number) {
      const gap = Date.now() - lastPong;
      if (gap >= tt.frozen) updateStatus("frozen");
      else if (gap >= tt.warning) updateStatus("warning");
      else if (gap >= tt.degraded) updateStatus("degraded");
    },

    getStatus: () => status,
    getRTT: () => rtt,
    getFirstDegradedAt: () => firstDegradedAt,
    getLastLoop: () => lastLoop,

    destroy() {
      status = "healthy";
      rtt = 0;
      firstDegradedAt = null;
      lastLoop = null;
    },
  };
}

// ─── Server-Side Probe ────────────────────────────────────────────────────────

export type TransportProbeServerConfig = {
  thresholds: VitalThresholds;
  onClientFrozen?: (clientId: string, frozenFor: number) => void;
  onClientRecovered?: (clientId: string) => void;
};

export type TransportProbeServerAPI = {
  onClientPing: (clientId: string, pingTime: number) => void;
  onClientStateSent: (clientId: string) => void;
  checkAllClients: () => void;
  getClientLiveness: (clientId: string) => ClientLiveness | undefined;
  getAllClients: () => ClientLiveness[];
  isFrozen: (clientId: string) => boolean;
  removeClient: (clientId: string) => void;
  destroy: () => void;
};

export function createTransportProbeServer(
  config: TransportProbeServerConfig,
): TransportProbeServerAPI {
  const tt = config.thresholds.transport;
  const clients = new Map<string, ClientLiveness>();

  function evaluateClient(liveness: ClientLiveness): void {
    const now = Date.now();
    const gap = now - liveness.lastPing;
    const prevStatus = liveness.status;

    if (gap >= tt.frozen) {
      liveness.status = "frozen";
      if (!liveness.frozenSince) liveness.frozenSince = now;
      if (prevStatus !== "frozen") {
        config.onClientFrozen?.(liveness.clientId, gap);
      }
    } else if (gap >= tt.warning) {
      liveness.status = "warning";
      if (prevStatus === "frozen") {
        liveness.frozenSince = undefined;
        config.onClientRecovered?.(liveness.clientId);
      }
    } else if (gap >= tt.degraded) {
      liveness.status = "degraded";
      if (prevStatus === "frozen") {
        liveness.frozenSince = undefined;
        config.onClientRecovered?.(liveness.clientId);
      }
    } else {
      if (prevStatus === "frozen") {
        liveness.frozenSince = undefined;
        config.onClientRecovered?.(liveness.clientId);
      }
      liveness.status = "healthy";
    }
  }

  return {
    onClientPing(clientId: string, pingTime: number) {
      const existing = clients.get(clientId);
      if (existing) {
        existing.lastPing = pingTime;
        evaluateClient(existing);
      } else {
        clients.set(clientId, {
          clientId,
          lastPing: pingTime,
          lastSent: 0,
          status: "healthy",
        });
      }
    },

    onClientStateSent(clientId: string) {
      const c = clients.get(clientId);
      if (c) c.lastSent = Date.now();
    },

    checkAllClients() {
      for (const liveness of clients.values()) {
        evaluateClient(liveness);
      }
    },

    getClientLiveness: (id) => clients.get(id),
    getAllClients: () => [...clients.values()],
    isFrozen: (id) => clients.get(id)?.status === "frozen",
    removeClient: (id) => {
      clients.delete(id);
    },

    destroy() {
      clients.clear();
    },
  };
}
```

```typescript
// src/vitals/mod.ts
import type {
  FreezeTimeline,
  VitalAlert,
  VitalLayer,
  VitalsConfig,
  VitalsSnapshot,
  VitalThresholds,
} from "./types.ts";
import { DEFAULT_HEARTBEAT_INTERVAL, DEFAULT_THRESHOLDS } from "./types.ts";
import { createLoopProbe, type LoopProbeAPI } from "./loop-probe.ts";
import {
  createTransportProbeServer,
  type TransportProbeServerAPI,
} from "./transport-probe.ts";
import { detectCascadeOrigin, evaluateHints } from "./hints.ts";
import { generateCorrelationId } from "../error.ts";

export type VitalsSystem = {
  loopProbe: LoopProbeAPI;
  serverTransport: TransportProbeServerAPI;
  checkAndAlert: () => void;
  getEndpointData: () => VitalsEndpointData;
  getLoopVitalsForPong: () => unknown;
  formatTimelineSummary: () => string;
  destroy: () => void;
};

export type VitalsEndpointData = {
  server: { loop: ReturnType<LoopProbeAPI["getVitals"]> };
  clients: Array<
    { id: string; status: string; rtt?: number; frozenFor?: number }
  >;
};

export function resolveThresholds(
  custom?: Partial<VitalThresholds>,
): VitalThresholds {
  if (!custom) return { ...DEFAULT_THRESHOLDS };
  return {
    render: custom.render ?? DEFAULT_THRESHOLDS.render,
    transport: custom.transport ?? DEFAULT_THRESHOLDS.transport,
    loop: custom.loop ?? DEFAULT_THRESHOLDS.loop,
    queue: custom.queue ?? DEFAULT_THRESHOLDS.queue,
  };
}

export function createVitalsSystem(config: VitalsConfig): VitalsSystem {
  const thresholds = resolveThresholds(config.thresholds);
  const hintsEnabled = config.hints !== false;
  const onAlert = config.onVitalAlert;

  const loopProbe = createLoopProbe(thresholds);
  const serverTransport = createTransportProbeServer({
    thresholds,
    onClientFrozen: (clientId, frozenFor) => {
      fireAlert("transport", "frozen", frozenFor, thresholds.transport.frozen);
    },
    onClientRecovered: (clientId) => {
      fireAlert("transport", "recovered", 0, 0);
    },
  });

  let checkTimer: ReturnType<typeof setInterval> | null = null;

  function fireAlert(
    layer: VitalLayer,
    status: VitalAlert["status"],
    measured: number,
    threshold: number,
  ) {
    if (!onAlert) return;
    const snap = buildSnapshot();
    const hint = hintsEnabled ? evaluateHints(snap, thresholds) : null;
    onAlert({
      id: generateCorrelationId(),
      layer,
      status,
      duration: measured,
      measured,
      threshold,
      hint,
      ts: Date.now(),
    });
  }

  function buildSnapshot(): VitalsSnapshot {
    const loopVitals = loopProbe.getVitals();
    return {
      render: {
        status: "healthy", // server-side doesn't know render state directly
        measured: 0,
        lastActionBefore: null,
        firstDegradedAt: null,
        visible: true,
      },
      transport: {
        status: "healthy",
        measured: 0,
        firstDegradedAt: null,
      },
      loop: {
        ...loopVitals,
        status: loopProbe.getStatus(),
        firstDegradedAt: loopProbe.getFirstDegradedAt(),
      },
    };
  }

  function checkAndAlert() {
    // Check loop status
    const loopStatus = loopProbe.getStatus();
    if (loopStatus !== "healthy") {
      const loopVitals = loopProbe.getVitals();
      const measured = Math.max(
        loopVitals.lastReduceTime,
        loopVitals.queueDepth,
      );
      const threshold = loopStatus === "frozen"
        ? thresholds.loop.frozen
        : loopStatus === "warning"
        ? thresholds.loop.warning
        : thresholds.loop.degraded;
      fireAlert("loop", loopStatus, measured, threshold);
    }

    // Check all client transports
    serverTransport.checkAllClients();
  }

  return {
    loopProbe,
    serverTransport,
    checkAndAlert,

    getEndpointData(): VitalsEndpointData {
      return {
        server: { loop: loopProbe.getVitals() },
        clients: serverTransport.getAllClients().map((c) => ({
          id: c.clientId,
          status: c.status,
          frozenFor: c.frozenSince ? Date.now() - c.frozenSince : undefined,
        })),
      };
    },

    getLoopVitalsForPong() {
      return loopProbe.getVitals();
    },

    formatTimelineSummary(): string {
      const snap = buildSnapshot();
      const loopVitals = loopProbe.getVitals();
      const hint = hintsEnabled ? evaluateHints(snap, thresholds) : null;
      const parts = [`[vitals:summary]`];
      parts.push(`loop: ${loopProbe.getStatus()}`);
      if (loopVitals.queueDepth > 0) {
        parts.push(`queue=${loopVitals.queueDepth}`);
      }
      if (loopVitals.lastReduceTime > 0) {
        parts.push(`reduce=${Math.round(loopVitals.lastReduceTime)}ms`);
      }
      if (hint) {
        parts.push(`| cause(${hint.severity}): ${hint.cause}`);
        parts.push(`| fix: ${hint.suggestion}`);
      }
      return parts.join(" ");
    },

    destroy() {
      if (checkTimer) clearInterval(checkTimer);
      loopProbe.reset();
      serverTransport.destroy();
    },
  };
}

// Re-export key types and probes for public API
export type {
  ClientLiveness,
  LoopVitals,
  VitalAlert,
  VitalHint,
  VitalsConfig,
  VitalsSnapshot,
  VitalStatus,
} from "./types.ts";
export { DEFAULT_HEARTBEAT_INTERVAL, DEFAULT_THRESHOLDS } from "./types.ts";
export { createLoopProbe } from "./loop-probe.ts";
export { createRenderProbe } from "./render-probe.ts";
export {
  createTransportProbeClient,
  createTransportProbeServer,
} from "./transport-probe.ts";
export {
  classifySeverity,
  detectCascadeOrigin,
  evaluateHints,
} from "./hints.ts";
```

```typescript
  vitals(
    layer: "render" | "transport" | "loop",
    status: string,
    measured: number,
    threshold: number,
    hint?: { cause: string; suggestion: string; severity: string },
  ): void {
    const msg = hint
      ? `[vitals:${layer}] ${status} ${Math.round(measured)}ms (threshold: ${threshold}ms) | cause(${hint.severity}): ${hint.cause} | fix: ${hint.suggestion}`
      : `[vitals:${layer}] ${status} ${Math.round(measured)}ms (threshold: ${threshold}ms)`;
    const entry: LogEntry = {
      ts: now(),
      lvl: status === "frozen" ? "warn" : "perf",
      cat: `vitals:${layer}`,
      msg,
      data: { layer, status, measured: Math.round(measured), threshold, hint },
    };
    this.write(this.path("perf"), entry);
    this.write(this.path("debug"), entry);
    if (this.cfg.console) printConsole(entry);
  }

  vitalsSummary(summary: string): void {
    const entry: LogEntry = {
      ts: now(),
      lvl: "info",
      cat: "vitals:summary",
      msg: summary,
    };
    this.write(this.path("app"), entry);
    this.write(this.path("debug"), entry);
    if (this.cfg.console) printConsole(entry);
  }
```
